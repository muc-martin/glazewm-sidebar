cmake_minimum_required(VERSION 3.24)
project(NativeSidebar VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Native Sidebar currently requires Windows and MSVC.")
endif()
add_executable(native-sidebar WIN32 src/main.cpp)
target_compile_features(native-sidebar PRIVATE cxx_std_17)
target_include_directories(native-sidebar PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(native-sidebar PRIVATE _WIN32_WINNT=0x0A00)
set_property(TARGET native-sidebar PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_options(native-sidebar PRIVATE /W4 /EHsc $<$<CONFIG:Release>:/O1>)
target_link_options(native-sidebar PRIVATE $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF> $<$<CONFIG:Release>:/INCREMENTAL:NO>)
target_link_libraries(native-sidebar PRIVATE user32 gdi32 winhttp shell32)
install(TARGETS native-sidebar RUNTIME DESTINATION .)
install(FILES sidebar.ini README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
install(DIRECTORY docs/ DESTINATION docs)
install(DIRECTORY packaging/ DESTINATION . FILES_MATCHING PATTERN "*.ps1" PATTERN "*.cmd")
enable_testing()
add_test(NAME installer-lifecycle COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass
  -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/Installer.Tests.ps1" -Binary "$<TARGET_FILE:native-sidebar>")
